-- CloutHub Key System (Correct Key with Dash & Auto Save 24h)
local CoreGui = game:GetService("CoreGui")

-- Tên file lưu key trong thư mục workspace của Executor
local saveFileName = "CloutHub_KeyData.json"
local isKeySaved = false

-- Kiểm tra file lưu key 24h
if writefile and readfile and isfile then
    if isfile(saveFileName) then
        local success, data = pcall(function()
            local HttpService = game:GetService("HttpService")
            return HttpService:JSONDecode(readfile(saveFileName))
        end)
        if success and data and data.expiryTime and os.time() < data.expiryTime then
            isKeySaved = true
        end
    end
end

-- Nếu đã lưu key và còn hạn, tải thẳng script chính
if isKeySaved then
    loadstring(game:HttpGet("https://raw.githubusercontent.com/ClouthubOnTop/Loader/main/main.lua"))()
    return
end

-- Hiệu ứng lắc mượt mà khi sai key
local function shakeFrame()
    local originalPos = UDim2.new(0.5, -200, 0.5, -135)
    local intensity = 9
    
    for i = 1, 3 do
        TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, intensity, 0, 0)}):Play()
        task.wait(0.04)
        TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, -intensity, 0, 0)}):Play()
        task.wait(0.04)
    end
    TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos}):Play()
end

-- Key chính xác (có chứa dấu gạch ngang)
local correctKey = "CloutHubKeyJskwoeih - Kalapwowkeloekeej"

-- Xử lý nút Get Key
GetKeyBtn.MouseButton1Click:Connect(function()
    local link = "https://link4m.org/2JkDl"
    if setclipboard then
        setclipboard(link)
        ErrorMsg.TextColor3 = Color3.fromRGB(0, 255, 120)
        ErrorMsg.Text = "Link copied to clipboard!"
    else
        ErrorMsg.TextColor3 = Color3.fromRGB(255, 215, 0)
        ErrorMsg.Text = "Your executor does not support auto-copy."
    end
end)

-- Xử lý nút Check Key
CheckBtn.MouseButton1Click:Connect(function()
    local enteredKey = TextBox.Text
    
    if enteredKey == correctKey then
        ErrorMsg.TextColor3 = Color3.fromRGB(0, 255, 120)
        ErrorMsg.Text = "Valid Key! Loading script..."
        
        -- Lưu thời hạn 24 giờ vào file
        if writefile then
            pcall(function()
                local HttpService = game:GetService("HttpService")
                local data = { expiryTime = os.time() + 86400 }
                writefile(saveFileName, HttpService:JSONEncode(data))
            end)
        end
        
        task.wait(0.8)
        
        local info = TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.In)
        TweenService:Create(MainFrame, info, {Size = UDim2.new(0, 0, 0, 0), BackgroundTransparency = 1}):Play()
        task.wait(0.3)
        
        ScreenGui:Destroy()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/ClouthubOnTop/Loader/main/main.lua"))()
    else
        ErrorMsg.TextColor3 = Color3.fromRGB(255, 70, 70)
        ErrorMsg.Text = "Invalid Key! Please try again."
        TextBox.Text = "" -- Xóa chữ khi nhập sai
        shakeFrame()
    end
end)
